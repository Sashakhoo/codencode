What a for loop really does

whatIn plain words

Anything you can loop over is an iterable. Behind every for loop, Python asks it for an iterator with iter(), then calls next() again and again until it raises StopIteration.

likeEveryday analogy

A ticket dispenser at the bank: each pull (next) gives you one number. It doesn't print all 500 tickets at once, and it tells you when the roll is empty.

rulesTheory: the key rules

  • iter(x) gets an iterator, next(it) gets one item
  • StopIteration ends the loop
  • Your class becomes iterable with __iter__ + __next__
  • An iterator remembers where it is: one item at a time
syntax at a glance
it = iter(["kopi", "teh"])
next(it)   # "kopi"
next(it)   # "teh"
next(it)   # StopIteration

Common mistakes

  • Looping over the same iterator twice: the second loop gets nothing.
  • Calling next() without handling StopIteration.

print(next(it))
print(next(it))
try:
    next(it)               # 3. nothing left -> StopIteration ends the loop
except StopIteration:
    print("StopIteration: loop ends here")


class Countdown:
    """Your own iterator: counts down to 1."""

    def __init__(self, start: int):
        self.current = start

    def __iter__(self):
        return self

    def __next__(self) -> int:
        if self.current < 1:
            raise StopIteration
        self.current -= 1
        return self.current + 1


print(list(Countdown(5)))
for n in Countdown(3):
    print("launch in", n)

Generators: functions that pause and resume

whatIn plain words

A function with yield instead of return is a generator. Each time you ask for a value it runs until the next yield, hands the value out, and pauses, remembering where it was. It makes values only when needed.

likeEveryday analogy

A roti canai man who makes each roti when you order it, not 1,000 in the morning that go cold.

rulesTheory: the key rules

  • yield value hands out a value and pauses
  • Calling the function returns a generator object; nothing runs yet
  • (x for x in …): generator expression, round brackets
  • One pass only; make a new one to start again

def countdown(n: int):
    """A generator: same Countdown as before, in 4 lines."""
    while n > 0:
        yield n            # pause here, hand out n, resume on next()
        n -= 1


g = countdown(3)
print(type(g).__name__)    # a generator object: nothing has run yet
print(next(g), next(g), next(g))


def queue_numbers(prefix: str, start: int = 1):
    """Endless ticket numbers for a clinic or bank queue."""
    n = start
    while True:
        yield f"{prefix}{n:03d}"
        n += 1


tickets = queue_numbers("A")
print([next(tickets) for _ in range(4)])

# generator expression: like a list comprehension, but with ( )
squares = (x * x for x in range(1_000_000))     # no big list in memory
print(sum(squares))

Streaming: process files bigger than your memory

whatIn plain words

Reading a whole file into a list puts every line in memory at once. A generator that yields one line at a time uses almost no memory, whether the file has 1,000 lines or 100 million.

likeEveryday analogy

Drinking from a tap instead of filling a swimming pool first. You get the same water without needing the pool.

rulesTheory: the key rules

  • for line in f:, since files are already iterators
  • Avoid .read() and .readlines() on big files
  • Chain steps: read → filter → parse → count
  • tracemalloc measures memory use

stream_log.py
import random
import tracemalloc
from pathlib import Path

# 1. make a big log file (200,000 lines) to practise on
log = Path("server.log")
random.seed(1)
with log.open("w") as f:
    for i in range(200_000):
        level = random.choice(["INFO"] * 9 + ["ERROR"])
        f.write(f"2026-10-21 09:{i % 60:02d}:00 {level} GET /page/{i % 50}\n")


def count_errors_list(path: Path) -> int:
    lines = path.read_text().splitlines()            # whole file in memory
    return len([line for line in lines if " ERROR " in line])


def read_lines(path: Path):
    with path.open() as f:
        for line in f:                               # one line at a time
            yield line


def count_errors_stream(path: Path) -> int:
    return sum(1 for line in read_lines(path) if " ERROR " in line)


for fn in (count_errors_list, count_errors_stream):
    tracemalloc.start()
    result = fn(log)
    peak = tracemalloc.get_traced_memory()[1] / 1_000_000
    tracemalloc.stop()
    print(f"{fn.__name__:<20} errors={result:,}  peak memory={peak:6.2f} MB")

real lifeBank statement analyser: stream, filter, total by merchant

Your bank lets you export a year of transactions. A chain of small generators reads, filters to one month, and converts amounts, so the same code works on 2,000 rows or 2 million.

statement.py
"""Bank statement analyser: stream a big CSV, total spending by merchant."""
import csv
import random
from collections import Counter
from pathlib import Path

# 1. a year of fake transactions (in real life: your bank's CSV export)
MERCHANTS = ["GrabFood", "Shopee", "TNB", "Petronas", "Tesco", "Netflix",
             "Watsons", "Starbucks"]
path = Path("statement.csv")
random.seed(7)
with path.open("w", newline="") as f:
    w = csv.writer(f)
    w.writerow(["date", "merchant", "amount"])
    for i in range(1_800):        # works the same on 1,000,000 rows
        w.writerow([f"2026-{i % 12 + 1:02d}-{i % 28 + 1:02d}",
                    random.choice(MERCHANTS), round(random.uniform(3, 80), 2)])


# 2. a pipeline of generators: read -> filter -> transform
def rows(p: Path):
    with p.open(newline="") as f:
        yield from csv.DictReader(f)


def in_month(records, month: str):
    return (r for r in records if r["date"].startswith(month))


def amounts(records):
    for r in records:
        yield r["merchant"], float(r["amount"])


totals: Counter = Counter()
for merchant, amount in amounts(in_month(rows(path), "2026-10")):
    totals[merchant] += amount

print("October 2026: top merchants")
for merchant, total in totals.most_common(5):
    print(f"  {merchant:<10} RM{total:>10,.2f}")
print(f"  {'ALL':<10} RM{sum(totals.values()):>10,.2f}")

Decorators: wrap a function with extra behaviour

whatIn plain words

A decorator is a function that takes a function and returns a new one that does something extra: timing, logging, retrying, checking permissions. You apply it with @name above the def.

likeEveryday analogy

Gift-wrapping: the present inside is unchanged, but now it has a card (log) and a ribbon (timer). You can wrap any present the same way.

rulesTheory: the key rules

  • def deco(func): def wrapper(*args, **kwargs): …; return wrapper
  • @deco above def means f = deco(f)
  • Always use @functools.wraps(func)
  • Stacked decorators apply bottom-up

@timer and @log_calls on a report function

decorators.py
import functools
import time


def timer(func):
    """Decorator: print how long func takes."""
    @functools.wraps(func)                    # keep the original name/docstring
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)        # call the real function
        ms = (time.perf_counter() - start) * 1000
        print(f"[timer] {func.__name__} took {ms:.1f} ms")
        return result
    return wrapper


def log_calls(func):
    """Decorator: print every call with its arguments."""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        print(f"[log] {func.__name__}{args} {kwargs or ''}")
        return func(*args, **kwargs)
    return wrapper


@timer
@log_calls
def monthly_report(month: str, rows: int = 50_000) -> int:
    return sum(i % 7 for i in range(rows))


print(monthly_report("2026-10"))
print(monthly_report.__name__)                # still 'monthly_report'

@lru_cache: remember answers you already worked out

whatIn plain words

If a function always gives the same output for the same input, @lru_cache stores each answer the first time. Later calls with the same input return instantly, without doing the slow work again.

likeEveryday analogy

The regular customer: the first time, the auntie asks how you like your kopi. After that she remembers and starts making it as you sit down.

rulesTheory: the key rules

  • Use for pure functions: same input → same output
  • Great for slow lookups, API calls with fixed answers
  • .cache_info() shows hits and misses
  • Don't cache time, random numbers or live prices
syntax at a glance
from functools import lru_cache

@lru_cache(maxsize=256)
def postcode_to_city(code):
    ...  # slow lookup

Common mistakes

  • Caching a function whose answer should change (stock levels).
  • Arguments must be hashable: lists can't be cache keys.

Postcode lookups: 1.2s without cache, 0.4s with

cache_demo.py
import time
from functools import lru_cache


def slow_lookup(postcode: str) -> str:
    time.sleep(0.2)                        # pretend this is a slow API/database
    return {"80000": "Johor Bahru", "50450": "Kuala Lumpur"}.get(postcode, "?")


@lru_cache(maxsize=256)
def cached_lookup(postcode: str) -> str:
    return slow_lookup(postcode)


orders = ["80000", "50450", "80000", "80000", "50450", "80000"]

start = time.perf_counter()
print([slow_lookup(p) for p in orders])
print(f"no cache : {time.perf_counter() - start:.2f}s")

start = time.perf_counter()
print([cached_lookup(p) for p in orders])
print(f"lru_cache: {time.perf_counter() - start:.2f}s")
print(cached_lookup.cache_info())

Regex: describe the shape, find every match

whatIn plain words

A regular expression describes the shape of text: "01, then a digit, then 7 or 8 digits". Python's re finds every piece of text with that shape. match/case picks an action based on the shape of data.

likeEveryday analogy

Telling someone "look for anything that looks like a car plate: letters, a space, numbers" instead of listing every plate in Malaysia.

rulesTheory: the key rules

  • \d digit · \w letter/digit · + one or more · ? optional
  • re.findall gives all matches · re.search the first
  • Named groups: (?P<day>\d{2}) → m["day"]
  • match cmd.split(): case ["add", item]:

msg = ("Order for table 4 la. Call me 012-345 6789 or my wife +60198765432. "
       "Email receipt to aisyah@example.com. Total RM 13.25, paid 21/10/2026.")

PHONE = re.compile(r"(?:\+?6)?01\d[- ]?\d{3,4}[- ]?\d{4}")
EMAIL = re.compile(r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+")
MONEY = re.compile(r"RM\s?(\d+(?:\.\d{2})?)")
DATE = re.compile(r"(?P<d>\d{2})/(?P<m>\d{2})/(?P<y>\d{4})")

print("phones:", PHONE.findall(msg))
print("emails:", EMAIL.findall(msg))
print("amount:", float(MONEY.search(msg).group(1)))
m = DATE.search(msg)
print("date  :", f"{m['y']}-{m['m']}-{m['d']}")


def handle(command: str) -> str:
    match command.split():                 # structural pattern matching
        case ["add", item, qty] if qty.isdigit():
            return f"adding {qty} x {item}"
        case ["add", item]:
            return f"adding 1 x {item}"
        case ["show" | "list"]:
            return "showing the order"
        case _:
            return f"unknown command: {command!r}"


for c in ["add teh 2", "add kopi", "list", "delete everything"]:

real lifeSupplier invoices: pull out number, amount and due date

Invoices arrive as email text in different formats. Regex patterns find the invoice number, RM amount, issue and due dates and the phone number, skip non-invoices, and total what you owe.

invoice_parser.py
"""Read supplier invoices from email text and pull out the key fields."""
import re

EMAILS = [
    """From: billing@mupackaging.example
Invoice No: INV-2026-0142  Date: 15/10/2026
Item: 500 pcs kraft box   Amount: RM 1,250.00
Payment due 30/10/2026. Contact 07-355 1234.""",
    """Hi, attached is invoice #INV-2026-0157 dated 18/10/2026.
Total payable RM980.50 by 02/11/2026. WhatsApp 012-888 9999 if any issue.""",
    """Reminder: your Netflix plan renews on 25/10/2026.""",
]

INVOICE = re.compile(r"INV-\d{4}-\d{4}")
AMOUNT = re.compile(r"RM\s?([\d,]+\.\d{2})")
DATES = re.compile(r"\b(\d{2})/(\d{2})/(\d{4})\b")
PHONE = re.compile(r"\b0\d{1,2}-\d{3,4} ?\d{4}\b")


def parse(text: str) -> dict | None:
    inv = INVOICE.search(text)
    if not inv:
        return None                                   # not an invoice
    dates = [f"{y}-{m}-{d}" for d, m, y in DATES.findall(text)]
    amount = AMOUNT.search(text)
    return {
        "invoice": inv.group(),
        "amount": float(amount.group(1).replace(",", "")) if amount else None,
        "issued": dates[0] if dates else None,
        "due": dates[-1] if len(dates) > 1 else None,
        "phone": (PHONE.search(text) or [None])[0],
    }


records = [r for r in map(parse, EMAILS) if r]
for r in records:
    print(r)
print(f"\n{len(records)} invoices, total RM{sum(r['amount'] for r in records):,.2f}")

def retry(times: int = 3):
    def decorator(func):

                    return func(*args, **kwargs)

                    time.sleep(attempt)
        return wrapper
    return decorator

A test is a function that checks your code is right

whatIn plain words

A test calls your function with a known input and asserts the output is what it should be. pytest finds every test_* function and runs them all in one command, so you know instantly if a change broke something.

likeEveryday analogy

A checklist the pilot runs before every flight. It takes seconds, and it catches the mistake before take-off, not in the air.

rulesTheory: the key rules

  • File test_*.py, function test_*()
  • assert actual == expected
  • Test boundaries: 79, 80, 81
  • Red → green → refactor

Fixtures set things up; parametrize multiplies tests

whatIn plain words

A fixture builds the same starting object for every test that asks for it by name, with no copy-pasted setup. parametrize runs one test function with many input/expected pairs.

likeEveryday analogy

A fixture is a clean mixing bowl handed to every baker. Parametrize is one recipe tested with 5 oven temperatures.

rulesTheory: the key rules

  • @pytest.fixture → use it as a test parameter
  • Each test gets a fresh copy
  • @pytest.mark.parametrize("a, b", [(1, 2), …])
  • Compare money with pytest.approx
syntax at a glance
@pytest.fixture
def cart():
    return Cart()

@pytest.mark.parametrize("voucher, expected", [(0, 39.9), (5, 34.9)])
def test_voucher(cart, voucher, expected): ...

Common mistakes

  • Tests that depend on each other's leftovers: fixtures avoid this.
  • 0.1 + 0.2 == 0.3 is False. Use approx.

A shopping cart: 1 fixture, 6 tests

test_cart.py
# Save as test_cart.py and run:  pytest -v test_cart.py
import pytest


class Cart:
    def __init__(self):
        self.items: dict[str, tuple[float, int]] = {}

    def add(self, name: str, price: float, qty: int = 1) -> None:
        if qty < 1:
            raise ValueError("qty must be at least 1")
        _, old_qty = self.items.get(name, (price, 0))
        self.items[name] = (price, old_qty + qty)

    def total(self, voucher: float = 0.0) -> float:
        subtotal = sum(p * q for p, q in self.items.values())
        return round(max(subtotal - voucher, 0), 2)


@pytest.fixture

Mocking: test API code without the internet

whatIn plain words

Tests must be fast and give the same answer every time, so they shouldn't call real APIs. A mock replaces requests.get with a fake that returns fixed data. pytest.raises checks that your code raises the right error.

likeEveryday analogy

A flight simulator. Pilots practise engine failures safely, on demand, without a real plane.

rulesTheory: the key rules

  • patch("module.requests.get", return_value=fake)
  • side_effect=ConnectionError fakes a network failure
  • with pytest.raises(ValueError, match="…")
  • Patch where it's used, not where it's defined

A currency converter tested with no network at all

test_rates.py
# Save as test_rates.py and run:  pytest -v test_rates.py
from unittest.mock import Mock, patch

import pytest
import requests


def get_rate(base: str, target: str) -> float:
    resp = requests.get("https://api.frankfurter.app/latest",
                        params={"from": base, "to": target}, timeout=10)
    resp.raise_for_status()
    return resp.json()["rates"][target]


def convert(amount: float, base: str, target: str) -> float:
    if amount < 0:
        raise ValueError("amount cannot be negative")
    return round(amount * get_rate(base, target), 2)


def test_convert_uses_rate():
    fake = Mock()
    fake.json.return_value = {"rates": {"USD": 0.21}}
    with patch("requests.get", return_value=fake) as mocked:   # no internet used
        assert convert(100, "MYR", "USD") == 21.0
    mocked.assert_called_once()


def test_negative_amount_rejected():
    with pytest.raises(ValueError, match="negative"):
        convert(-5, "MYR", "USD")


def test_api_down_raises():
    with patch("requests.get", side_effect=requests.ConnectionError):
        with pytest.raises(requests.ConnectionError):
            get_rate("MYR", "USD")

# Real-life: test a payroll calculator before it pays real people.
# Save as test_payroll.py and run:  pytest -v test_payroll.py
import pytest


def overtime_pay(hourly: float, hours: float) -> float:
    """Normal rate up to 8 hours/day, 1.5x after that."""
    if hourly <= 0 or hours < 0:
        raise ValueError("hourly must be > 0 and hours >= 0")
    normal = min(hours, 8) * hourly
    extra = max(hours - 8, 0) * hourly * 1.5
    return round(normal + extra, 2)


def net_salary(gross: float, epf_rate: float = 0.11) -> float:
    """Gross minus employee EPF (simplified rate for class)."""
    return round(gross * (1 - epf_rate), 2)


@pytest.mark.parametrize("hours, expected", [
    (0, 0.0), (4, 40.0), (8, 80.0), (10, 110.0), (12, 140.0),
])
def test_overtime(hours, expected):
    assert overtime_pay(10, hours) == expected


@pytest.mark.parametrize("hourly, hours", [(0, 8), (-5, 8), (10, -1)])
def test_bad_input(hourly, hours):
    with pytest.raises(ValueError):
        overtime_pay(hourly, hours)


def test_net_salary_default_epf():
    assert net_salary(3000) == pytest.approx(2670.00)


def test_net_salary_custom_epf():
    assert net_salary(3000, epf_rate=0.09) == pytest.approx(2730.00)
