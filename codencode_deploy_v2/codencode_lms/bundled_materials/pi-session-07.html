What makes a good coding prompt?

whatIn plain words

Claude can only use what you tell it. A good prompt gives five things: context (project, language), the task (one job), constraints (rules, edge cases), an example (input → output), and the output you want back.

likeEveryday analogy

Briefing a new staff member: "This is our POS (context). Add a split-bill button (task). Round to the sen, no new hardware (constraints). Like this receipt (example). Show me before Friday (output)."

rulesTheory: the key rules

  • Context: what and where
  • Task: one clear job per prompt
  • Constraints + example: remove guessing
  • Output: always ask for tests + an explanation

Build the 5-part prompt with a function

prompt_builder.py
def build_prompt(context: str, task: str, constraints: list[str],
                 example: str = "", output: str = "") -> str:
    """Assemble a 5-part coding prompt you can paste into Claude."""
    parts = [f"## Context\n{context}", f"## Task\n{task}",
             "## Constraints\n" + "\n".join(f"- {c}" for c in constraints)]
    if example:
        parts.append(f"## Example\n{example}")
    if output:
        parts.append(f"## Output\n{output}")
    return "\n\n".join(parts)


prompt = build_prompt(
    context="Kopitiam POS CLI in Python 3.12. Order class stores (name, price) items.",
    task="Add a split_bill(n) method that divides the total between n people.",
    constraints=["type hints and a docstring", "amounts in sen-accurate RM, 2 dp",
                 "leftover sen goes to the first person", "raise ValueError if n < 1",
                 "no new libraries"],
    example="RM10.00 split 3 ways -> [3.34, 3.33, 3.33]",
    output="The method, 3 pytest tests, then explain the rounding in 2 sentences.",
)
print(prompt)
print(f"\n({len(prompt.split())} words)")

Pack your project into one paste-ready file

context_pack.py
"""Pack your project's code into one Markdown file to paste into Claude.
Skips folders you never want to share and hides secret-looking lines."""
import re
from pathlib import Path

PROJECT = Path("my_project")
SKIP_DIRS = {".venv", "__pycache__", ".git", "data"}
SECRET = re.compile(r"(?i)(api[_-]?key|password|secret|token)\s*[:=]\s*\S+")

# demo only: create a tiny project if you don't have one yet
if not PROJECT.exists():
    (PROJECT / "src").mkdir(parents=True)
    (PROJECT / "src" / "models.py").write_text(
        "class Order:\n    def __init__(self, table):\n        self.table = table\n")
    (PROJECT / "src" / "config.py").write_text(
        'API_KEY = "sk-live-9f8e7d"\nSHOP = "Orkid Kopitiam"\n')
    (PROJECT / ".venv").mkdir()
    (PROJECT / ".venv" / "huge.py").write_text("# should be skipped\n")


def project_files(root: Path):
    for path in sorted(root.rglob("*.py")):
        if not SKIP_DIRS & set(path.parts):
            yield path


chunks = []
for path in project_files(PROJECT):
    code = SECRET.sub(lambda m: f"{m.group(1)} = <REDACTED>", path.read_text())
    chunks.append(f"### {path.relative_to(PROJECT)}\n```python\n{code}```")
    print("included:", path.relative_to(PROJECT))

Path("context.md").write_text("\n\n".join(chunks))
print("\n" + Path("context.md").read_text())

The rules: no secrets, and understand every line

whatIn plain words

Never paste secrets (API keys, passwords) or personal data (IC numbers, real phone numbers) into any AI tool. Run every suggestion, read every line, and only keep code you can explain. You own the result.

likeEveryday analogy

Copying a classmate's homework: it might be right, but if you can't explain it, the exam will show it.

rulesTheory: the key rules

  • Fake data only: no real customers
  • Run → read → change one thing → explain
  • One feature per prompt; commit when green
  • Note "Built with Claude" in your README
syntax at a glance
# After AI gives you code, always:
# 1. Run it.
# 2. Read every line.
# 3. Change one thing on purpose; predict what breaks.
# 4. If you can't explain it, you don't own it yet.

Common mistakes

  • Accepting a 200-line answer without running it.
  • Letting AI edit a test just so it passes.

Scan text for secrets before you paste it

secret_scan.py
"""Check text BEFORE you paste it into any AI tool."""
import re

CHECKS = {
    "API key":       r"\b(sk|pk|rk)[-_](live|test)?[-_]?[A-Za-z0-9]{6,}\b",
    "password":      r"(?i)\bpass(word)?\s*[:=]\s*\S+",
    "MyKad number":  r"\b\d{6}-\d{2}-\d{4}\b",
    "card number":   r"\b(?:\d{4}[ -]?){3}\d{4}\b",
    "phone number":  r"(?:\+?6)?01\d[- ]?\d{3,4}[- ]?\d{4}",
}


def scan(text: str) -> list[str]:
    """Return a warning for every risky thing found."""
    found = []
    for label, pattern in CHECKS.items():
        for m in re.finditer(pattern, text):
            hidden = m.group()[:3] + "*" * (len(m.group()) - 3)
            found.append(f"{label:<13} {hidden}")
    return found


paste = """Hi Claude, my code fails:
API_KEY = "sk-live-AbC123xyz789"
customer = {"ic": "900101-01-1234", "phone": "012-345 6789"}
db password: hunter2
"""
warnings = scan(paste)
print("SAFE TO PASTE" if not warnings else f"STOP - {len(warnings)} problem(s):")
for w in warnings:
    print("  " + w)
print("\nclean text:", "SAFE" if not scan("def total(items): return sum(items)") else "!")

real lifeYour personal prompt library

Good prompts are worth saving. Store your best review, debug and test prompts as templates, fill in the blanks with one function call, and keep them in a prompts/ folder you reuse on every project.

prompt_library.py
"""Your personal prompt library: reuse your best prompts with one function call."""
from pathlib import Path
from string import Template

LIBRARY = {
    "review": Template(
        "Review this $language code as a senior developer. List at most "
        "$max_issues issues, most important first: the line, the problem, a fix. "
        "Don't rewrite everything.\n\n$code"),
    "debug": Template(
        "Expected: $expected\nActual: $actual\nTraceback:\n$traceback\n\n"
        "Don't fix it yet. Tell me the likely cause and one print or breakpoint "
        "to confirm it."),
    "tests": Template(
        "Write pytest tests for this function. Cover a normal case, each "
        "boundary and each error. Use parametrize. One comment per test.\n\n$code"),
}


def prompt(name: str, **fields: str) -> str:
    """Fill a saved prompt. Missing fields raise KeyError so you notice."""
    return LIBRARY[name].substitute(**fields)


code = "def grade(m):\n    return 'A' if m > 80 else 'F'"
p = prompt("review", language="Python", max_issues="3", code=code)
print(p, "\n" + "-" * 40)
print(prompt("debug", expected="grade(80) == 'A'", actual="'F'",
             traceback="(no crash, wrong answer)"))

Path("prompts").mkdir(exist_ok=True)
for name, template in LIBRARY.items():
    Path("prompts", f"{name}.txt").write_text(template.template)
print("-" * 40, "\nsaved:", sorted(p.name for p in Path("prompts").iterdir()))

One loop's result: a feature with its tests

test_split_bill.py
# Feature built in one Claude loop: split a bill fairly to the sen.
# Run:  pytest -v test_c7_04_feature.py
import pytest


def split_bill(total: float, people: int) -> list[float]:
    """Split total between people. Leftover sen go to the first people.

    Works in whole sen (integers) to avoid float errors like 0.1 + 0.2.
    """
    if people < 1:
        raise ValueError("people must be at least 1")
    sen = round(total * 100)
    base, leftover = divmod(sen, people)
    return [(base + (1 if i < leftover else 0)) / 100 for i in range(people)]


print(split_bill(10.00, 3))      # [3.34, 3.33, 3.33]


@pytest.mark.parametrize("total, people, expected", [
    (10.00, 3, [3.34, 3.33, 3.33]),
    (25.40, 2, [12.70, 12.70]),
    (0.05, 2, [0.03, 0.02]),
    (99.99, 1, [99.99]),
])
def test_split(total, people, expected):
    assert split_bill(total, people) == expected


def test_shares_add_up():
    assert round(sum(split_bill(123.45, 7)), 2) == 123.45


def test_zero_people():
    with pytest.raises(ValueError):
        split_bill(10, 0)

# AI's first draft: looked fine, but crashed on real data.
def class_average_v1(marks: dict[str, list[int]]) -> dict[str, float]:
    return {cls: round(sum(m) / len(m)) for cls, m in marks.items()}


data = {"5A": [88, 75, 91], "5B": [], "5C": [64, 70]}

try:
    print(class_average_v1(data))
except ZeroDivisionError as e:
    print("v1 crashed:", type(e).__name__, "-", e)

# Debug prompt we sent:  Expected / Actual / Traceback / "diagnose first"
# Claude's diagnosis: class 5B has no marks -> len(m) == 0.
# Second bug we spotted ourselves while reading: round() with no digits
# turns 84.67 into 85, which is wrong for a report.


def class_average_v2(marks: dict[str, list[int]]) -> dict[str, float | None]:
    """Average per class to 1 decimal. Empty classes give None."""
    return {cls: round(sum(m) / len(m), 1) if m else None
            for cls, m in marks.items()}


print("v2:", class_average_v2(data))

What a finished small project looks like: dataclasses for lines, a class for orders, SQLite for sales history, a receipt with SST, best sellers, and bad input rejected. Every class in the course appears here.

pos.py
"""Mini capstone: Kopitiam POS in one file (classes + SQLite + receipt)."""
import sqlite3
from dataclasses import dataclass

SST = 0.06
MENU = {"TT": ("Teh Tarik", 3.00), "KO": ("Kopi O", 2.50),
        "NL": ("Nasi Lemak", 6.50), "RC": ("Roti Canai", 2.50)}


@dataclass
class Line:
    name: str
    price: float
    qty: int

    @property
    def total(self) -> float:
        return self.price * self.qty


class Order:
    def __init__(self, table: int):
        self.table = table
        self.lines: list[Line] = []

    def add(self, code: str, qty: int = 1) -> None:
        if code not in MENU:
            raise KeyError(f"no menu item {code!r}")
        name, price = MENU[code]
        self.lines.append(Line(name, price, qty))

    def subtotal(self) -> float:
        return sum(line.total for line in self.lines)

    def receipt(self) -> str:
        rows = [f"{line.qty} x {line.name:<12} RM{line.total:>6.2f}" for line in self.lines]
        sst = self.subtotal() * SST
        rows += ["-" * 28, f"{'SST 6%':<16} RM{sst:>6.2f}",
                 f"{'TOTAL':<16} RM{self.subtotal() + sst:>6.2f}"]
        return f"Table {self.table}\n" + "\n".join(rows)


class Sales:
    def __init__(self, path: str = "pos.db"):
        self.conn = sqlite3.connect(path)
        self.conn.execute("CREATE TABLE IF NOT EXISTS sales (tbl INT, item TEXT, qty INT, amount REAL)")

    def save(self, order: Order) -> None:
        self.conn.executemany("INSERT INTO sales VALUES (?, ?, ?, ?)",
                              [(order.table, ln.name, ln.qty, ln.total) for ln in order.lines])
        self.conn.commit()

    def best_sellers(self) -> list[tuple[str, int]]:
        return self.conn.execute("SELECT item, SUM(qty) FROM sales GROUP BY item "
                                 "ORDER BY SUM(qty) DESC LIMIT 3").fetchall()


sales = Sales(":memory:")
for table, items in [(4, [("TT", 2), ("NL", 1)]), (7, [("KO", 3), ("RC", 2)]),
                     (2, [("TT", 1), ("RC", 1)])]:
    order = Order(table)
    for code, qty in items:
        order.add(code, qty)
    sales.save(order)
print(order.receipt())
print("\nbest sellers today:", sales.best_sellers())
try:
    Order(1).add("XX")
except KeyError as e:
    print("rejected:", e)

"""ship_check.py: run the 8-point checklist on a project folder."""
import re
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(sys.argv[1] if len(sys.argv) > 1 else ".")
results: list[tuple[bool, str]] = []


def check(ok: bool, label: str) -> None:
    results.append((ok, label))


readme = ROOT / "README.md"
text = readme.read_text() if readme.exists() else ""
check(all(h in text for h in ("## Install", "## Usage")), "README has Install + Usage")
check((ROOT / "requirements.txt").exists(), "requirements.txt exists")
gitignore = (ROOT / ".gitignore").read_text() if (ROOT / ".gitignore").exists() else ""
check(".env" in gitignore.split(), ".env is in .gitignore")

secret = re.compile(r"(sk|pk)[-_](live|test)[-_]?\w{6,}")
leaks = [p for p in ROOT.rglob("*.py") if ".venv" not in p.parts
         and secret.search(p.read_text(errors="ignore"))]
check(not leaks, f"no secrets in code {[str(p) for p in leaks] if leaks else ''}")
check(any(ROOT.rglob("test_*.py")), "has test files")

tests = subprocess.run([sys.executable, "-m", "pytest", "-q", str(ROOT)],
                       capture_output=True, text=True)
check(tests.returncode == 0, "pytest passes")
if shutil.which("ruff"):
    lint = subprocess.run(["ruff", "check", str(ROOT)], capture_output=True, text=True)
    check(lint.returncode == 0, "ruff check passes")

for ok, label in results:
    print(("  OK  " if ok else "  FIX ") + label)
passed = sum(ok for ok, _ in results)
print(f"\n{passed}/{len(results)} passed -", "SHIP IT" if passed == len(results) else "not yet")

"""demo.py: run your live demo as a script so nothing is forgotten on stage."""
import subprocess
import sys
import time

STEPS = [
    ("1. The problem", None),
    ("2. Normal use", [sys.executable, "-c", "print('Table 4: 2 x Teh Tarik  RM6.00')"]),
    ("3. Bad input is handled", [sys.executable, "-c",
                                 "print('error: argument table: invalid int value: four')"]),
    ("4. Tests pass", [sys.executable, "-c", "print('8 passed in 0.11s')"]),
]
SPEAKER_NOTES = {"1. The problem": "Hawker stalls still take orders on paper."}


def run_demo(pause: float = 0.0) -> None:
    for title, command in STEPS:
        print(f"\n=== {title}")
        if title in SPEAKER_NOTES:
            print(f"(say) {SPEAKER_NOTES[title]}")
        if command:
            out = subprocess.run(command, capture_output=True, text=True)
            print(out.stdout.strip() or out.stderr.strip())
        time.sleep(pause)         # set pause=2 on the day


run_demo()
print("\nReplace the commands with your real ones, e.g. "
      "[sys.executable, '-m', 'kopitiam.cli', 'show', '4']")

real liferelease.py: test, version, zip a clean copy

Submitting a project or handing it to a client: the script refuses to package if tests fail, bumps the version number, and zips everything except secrets and junk, then proves .env isn't inside.

release.py
"""release.py: check -> bump version -> zip a clean copy for submission."""
import subprocess
import sys
import zipfile
from datetime import date
from pathlib import Path

ROOT = Path("my_project")
EXCLUDE = {".venv", "__pycache__", ".git", ".pytest_cache"}
EXCLUDE_FILES = {".env"}

# demo only: create a small project if it doesn't exist
if not ROOT.exists():
    (ROOT / "src").mkdir(parents=True)
    (ROOT / "src" / "app.py").write_text("def add(a, b):\n    return a + b\n")
    (ROOT / "test_app.py").write_text(
        "import sys\nsys.path.insert(0, 'src')\nfrom app import add\n\n"
        "def test_add():\n    assert add(2, 3) == 5\n")
    (ROOT / ".env").write_text("API_KEY=secret\n")
    (ROOT / "README.md").write_text("# My Project\n")

# 1. tests must pass
result = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider"],
                        cwd=ROOT, capture_output=True, text=True)
print("tests:", result.stdout.strip().splitlines()[-1])
if result.returncode != 0:
    sys.exit("stopping: fix the tests first")

# 2. bump the version
version_file = ROOT / "VERSION"
major, minor = map(int, (version_file.read_text().strip() if version_file.exists()
                         else "1.0").split("."))
version = f"{major}.{minor + 1}"
version_file.write_text(version + "\n")

# 3. zip everything except secrets and junk
zip_name = Path(f"my_project_v{version}_{date(2026, 10, 21):%Y%m%d}.zip")
with zipfile.ZipFile(zip_name, "w", zipfile.ZIP_DEFLATED) as z:
    for path in sorted(ROOT.rglob("*")):
        if path.is_file() and not EXCLUDE & set(path.parts) and path.name not in EXCLUDE_FILES:
            z.write(path, path.relative_to(ROOT.parent))
with zipfile.ZipFile(zip_name) as z:
    names = z.namelist()
print(f"version {version} -> {zip_name.name}")
for n in names:
    print("  ", n)
print(".env included?", any(n.endswith(".env") for n in names))
