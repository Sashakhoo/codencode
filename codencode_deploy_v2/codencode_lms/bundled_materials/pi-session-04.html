Inheritance: a child class gets everything the parent has

whatIn plain words

Inheritance lets a new class reuse an existing one: class Manager(Employee). The child gets all the parent's attributes and methods, then adds or changes a few. super() calls the parent's version.

likeEveryday analogy

A new kopitiam branch copies the original's menu and recipes, then adds its own special dish. It doesn't rewrite the whole menu.

rulesTheory: the key rules

  • class Child(Parent): means Child IS A Parent
  • Child inherits every method it doesn't define
  • super().__init__(…) lets the parent set up its part
  • isinstance(obj, Parent) is True for children

class Employee:
    EPF_RATE = 0.11              # simplified for class

    def __init__(self, name: str, base_salary: float):
        self.name = name
        self.base_salary = base_salary

    def monthly_pay(self) -> float:
        return self.base_salary

    def epf(self) -> float:
        return round(self.monthly_pay() * self.EPF_RATE, 2)


class Manager(Employee):                     # Manager IS AN Employee
    def __init__(self, name: str, base_salary: float, allowance: float):
        super().__init__(name, base_salary)  # parent sets name + salary
        self.allowance = allowance           # child adds its own


sasha = Manager("Sasha", 6000, 800)
print(sasha.name, sasha.base_salary, sasha.allowance)
print("EPF (inherited method):", sasha.epf())
print(isinstance(sasha, Employee), Manager.__mro__[1].__name__)

Polymorphism: one call, the right behaviour for each object

whatIn plain words

A child can override a parent method by defining one with the same name. Then you can call s.monthly_pay() on a mixed list and each object runs its own version, with no if/elif on the type.

likeEveryday analogy

"Make the drink" means something different to the teh tarik maker and the juice stall, but the cashier gives the same instruction to both.

rulesTheory: the key rules

  • Same method name in the child = override
  • super().method() reuses the parent's result
  • Loop over mixed objects, call the same method
  • if type(x) == … chains usually mean you should override

class Employee:
    def __init__(self, name: str, base_salary: float):
        self.name = name
        self.base_salary = base_salary

    def monthly_pay(self) -> float:
        return self.base_salary

    def epf(self) -> float:
        return round(self.monthly_pay() * 0.11, 2)


class Manager(Employee):
    def __init__(self, name: str, base_salary: float, allowance: float):
        super().__init__(name, base_salary)
        self.allowance = allowance

    def monthly_pay(self) -> float:          # override + reuse the parent
        return super().monthly_pay() + self.allowance


class Intern(Employee):
    def monthly_pay(self) -> float:          # override: fixed allowance
        return 1200.0

    def epf(self) -> float:                  # override: no EPF
        return 0.0


staff = [Employee("Tan", 3500), Manager("Sasha", 6000, 800), Intern("Nurin", 0)]
print(f"{'Name':<8}{'Pay':>10}{'EPF':>9}{'Net':>10}")
for s in staff:                              # same call, right answer for each
    pay, epf = s.monthly_pay(), s.epf()
    print(f"{s.name:<8}{pay:>10.2f}{epf:>9.2f}{pay - epf:>10.2f}")

Composition: build objects from swappable parts

whatIn plain words

Composition means an object has other objects as parts: an Order has a payment method. Swap the part and the Order code doesn't change. Prefer it over deep inheritance.

likeEveryday analogy

A phone has a SIM card. You swap Celcom for Maxis without buying a new phone. The phone isn't a SIM; it has one.

rulesTheory: the key rules

  • "is a" → inherit · "has a" → compose
  • Pass the part into __init__
  • Parts share method names (pay()) so they're swappable
  • Inheritance deeper than 2 levels is a warning sign
syntax at a glance
class Order:
    def __init__(self, table, payment):
        self.payment = payment      # HAS a payment method

    def checkout(self):
        return self.payment.pay(self.total())

Common mistakes

  • class Order(CashPayment): an order is not a payment.
  • Creating the part inside __init__, so it can't be swapped.

class CashPayment:
    def pay(self, amount: float) -> str:
        return f"Cash RM{amount:.2f}"


class EWalletPayment:
    def __init__(self, provider: str):
        self.provider = provider

    def pay(self, amount: float) -> str:
        return f"{self.provider} e-wallet RM{amount:.2f}"


class CardPayment:
    def __init__(self, last4: str):
        self.last4 = last4

    def pay(self, amount: float) -> str:
        return f"Card ****{self.last4} RM{amount:.2f}"


class Order:
    """An Order HAS items and HAS a payment method (composition)."""

    def __init__(self, table: int, payment):
        self.table = table
        self.items: list[tuple[str, float]] = []
        self.payment = payment

    def add(self, name: str, price: float) -> None:
        self.items.append((name, price))

    def checkout(self) -> str:
        total = round(sum(p for _, p in self.items) * 1.06, 2)
        return f"Table {self.table} paid by {self.payment.pay(total)}"


for method in [CashPayment(), EWalletPayment("TNG"), CardPayment("4242")]:
    order = Order(4, method)          # swap the part, Order code never changes
    order.add("Nasi Lemak", 6.5)
    order.add("Teh Tarik", 3.0)
    print(order.checkout())

real lifeClass reminders by email, SMS and WhatsApp

Students choose how they want reminders. Each channel is a child class with its own send(); the reminder service has a set of notifiers and uses them polymorphically. It prints instead of sending, so it's safe to run.

reminders.py
"""Class reminders through each student's preferred channels.
(Prints instead of really sending, so it is safe to run.)"""


class Notifier:
    channel = "base"

    def send(self, to: str, message: str) -> None:
        raise NotImplementedError("each channel must implement send()")

    def clean(self, message: str) -> str:            # shared by every child
        return " ".join(message.split())


class EmailNotifier(Notifier):
    channel = "email"

    def send(self, to: str, message: str) -> None:
        print(f"[EMAIL -> {to}] Subject: Class reminder | {self.clean(message)}")


class SMSNotifier(Notifier):
    channel = "sms"
    LIMIT = 60                                       # SMS must be short

    def send(self, to: str, message: str) -> None:
        text = self.clean(message)
        if len(text) > self.LIMIT:
            text = text[: self.LIMIT - 3] + "..."
        print(f"[SMS   -> {to}] {text}")


class WhatsAppNotifier(Notifier):
    channel = "whatsapp"

    def send(self, to: str, message: str) -> None:
        print(f"[WA    -> {to}] Hi! {self.clean(message)}")


class ReminderService:
    """HAS a set of notifiers (composition) and uses them polymorphically."""

    def __init__(self, notifiers: list[Notifier]):
        self.notifiers = {n.channel: n for n in notifiers}

    def remind(self, student: dict, message: str) -> None:
        for channel in student["prefers"]:
            self.notifiers[channel].send(student[channel], message)


service = ReminderService([EmailNotifier(), SMSNotifier(), WhatsAppNotifier()])
students = [
    {"name": "Aisyah", "prefers": ["whatsapp"], "whatsapp": "+6012-0000001"},
    {"name": "Darren", "prefers": ["email", "sms"],
     "email": "darren@example.com", "sms": "+6012-0000002"},
]
msg = "Python Intermediate Class 4 starts   tomorrow 8pm on Zoom. Bring your laptop!"
for s in students:
    service.remind(s, msg)

SQLite: a real database in one file

whatIn plain words

A database stores data in tables that survive after your program closes. SQLite is a full database in a single file, built into Python. CRUD = Create, Read, Update, Delete: the four things every app does.

likeEveryday analogy

A filing cabinet (database) with labelled drawers (tables) and one card per customer (row). SQL is how you ask the clerk to file, find, change or shred a card.

rulesTheory: the key rules

  • INSERT create · SELECT read · UPDATE change · DELETE remove
  • sqlite3.connect("shop.db") creates the file if missing
  • Writes need conn.commit()
  • executemany inserts many rows at once

import sqlite3

conn = sqlite3.connect("kopitiam.db")      # creates the file if missing
conn.execute("DROP TABLE IF EXISTS orders")  # fresh start for this demo
conn.execute("""CREATE TABLE orders (
                  id INTEGER PRIMARY KEY,
                  tbl INTEGER, item TEXT, qty INTEGER)""")

# CREATE
conn.executemany("INSERT INTO orders (tbl, item, qty) VALUES (?, ?, ?)",
                 [(4, "Teh Tarik", 2), (4, "Nasi Lemak", 1), (7, "Milo Ais", 3)])
conn.commit()

# READ
for row in conn.execute("SELECT id, tbl, item, qty FROM orders"):
    print(row)

# UPDATE
conn.execute("UPDATE orders SET qty = ? WHERE id = ?", (3, 1))
# DELETE
conn.execute("DELETE FROM orders WHERE tbl = ?", (7,))
conn.commit()

print(conn.execute("SELECT * FROM orders").fetchall())
conn.close()

Never build SQL with f-strings

whatIn plain words

If you glue user input into SQL with an f-string, a user can type SQL code and change your query: that's SQL injection. Use ? placeholders and pass values separately; the database then treats them as plain data.

likeEveryday analogy

Writing a cheque: you fill in the amount box. With an f-string, it's like letting the customer write anywhere on the cheque, including "pay me everything".

rulesTheory: the key rules

  • ? placeholder for every value
  • Pass values as a tuple: (name,) for one value
  • Never use f-strings, + or % in SQL
  • This applies to every database, not just SQLite

An attack in action, then blocked

injection_demo.py
import sqlite3

conn = sqlite3.connect(":memory:")         # throwaway database in RAM
conn.execute("CREATE TABLE users (name TEXT, salary REAL)")
conn.executemany("INSERT INTO users VALUES (?, ?)",
                 [("Aisyah", 4200), ("Darren", 5100), ("Nurin", 3900)])

evil = "x' OR '1'='1"                      # what an attacker types in a search box

# UNSAFE: the input becomes part of the SQL itself
unsafe = f"SELECT * FROM users WHERE name = '{evil}'"
print("unsafe SQL:", unsafe)
print("unsafe result:", conn.execute(unsafe).fetchall())   # leaks every salary!

# SAFE: ? keeps the input as plain data
safe = conn.execute("SELECT * FROM users WHERE name = ?", (evil,)).fetchall()
print("safe result:", safe)                                # nothing matches
print("normal search:",
      conn.execute("SELECT * FROM users WHERE name = ?", ("Nurin",)).fetchall())

A repository class keeps all the SQL in one place

whatIn plain words

Wrap the database in a class (a repository) with plain methods like add() and by_table(). The rest of the app calls those methods and never writes SQL.

likeEveryday analogy

The clerk at the counter. You say "file this order"; you don't walk into the back room and open drawers yourself.

rulesTheory: the key rules

  • One class per table or topic: OrderRepository
  • row_factory = sqlite3.Row → rows act like dicts
  • ":memory:" gives a throwaway DB for tests
  • Commit inside the method that writes
syntax at a glance
class OrderRepository:
    def __init__(self, path="shop.db"):
        self.conn = sqlite3.connect(path)

    def add(self, tbl, item, qty, price): ...
    def by_table(self, tbl): ...
    def delete(self, order_id): ...

Common mistakes

  • SQL scattered across 10 files: impossible to change later.
  • Inheriting from sqlite3.Connection instead of having one.

OrderRepository: the app never sees SQL

repository.py
import sqlite3


class OrderRepository:
    """All the SQL lives here. The rest of the app never writes SQL."""

    def __init__(self, path: str = "kopitiam.db"):
        self.conn = sqlite3.connect(path)
        self.conn.row_factory = sqlite3.Row          # rows behave like dicts
        self.conn.execute("""CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY, tbl INTEGER, item TEXT,
            qty INTEGER, price REAL)""")

    def add(self, tbl: int, item: str, qty: int, price: float) -> int:
        cur = self.conn.execute(
            "INSERT INTO orders (tbl, item, qty, price) VALUES (?, ?, ?, ?)",
            (tbl, item, qty, price))
        self.conn.commit()
        return cur.lastrowid

    def by_table(self, tbl: int) -> list[sqlite3.Row]:
        return self.conn.execute(
            "SELECT * FROM orders WHERE tbl = ? ORDER BY id", (tbl,)).fetchall()

    def table_total(self, tbl: int) -> float:
        row = self.conn.execute(
            "SELECT COALESCE(SUM(qty * price), 0) FROM orders WHERE tbl = ?",
            (tbl,)).fetchone()
        return row[0]

    def delete(self, order_id: int) -> None:
        self.conn.execute("DELETE FROM orders WHERE id = ?", (order_id,))
        self.conn.commit()


repo = OrderRepository(":memory:")       # use a file path in real life
repo.add(4, "Teh Tarik", 2, 3.00)
oid = repo.add(4, "Roti Canai", 1, 2.50)
repo.add(4, "Nasi Lemak", 1, 6.50)
repo.delete(oid)
for r in repo.by_table(4):
    print(r["qty"], "x", r["item"], f"RM{r['qty'] * r['price']:.2f}")
print("table 4 total: RM", repo.table_total(4))

real lifeCustomer contact book that remembers between runs

A small business keeps customer contacts. Duplicate phone numbers are ignored, search works on name or company, and the data is still there tomorrow. Run it twice and the second run says "exists".

contacts.py
"""Customer contact book saved in SQLite. Run it twice: data is still there."""
import sqlite3
from pathlib import Path


class ContactBook:
    def __init__(self, path: Path):
        self.conn = sqlite3.connect(path)
        self.conn.row_factory = sqlite3.Row
        self.conn.execute("""CREATE TABLE IF NOT EXISTS contacts (
            id INTEGER PRIMARY KEY,
            name TEXT NOT NULL,
            phone TEXT UNIQUE NOT NULL,
            company TEXT)""")

    def add(self, name: str, phone: str, company: str = "") -> bool:
        cur = self.conn.execute(
            "INSERT OR IGNORE INTO contacts (name, phone, company) VALUES (?, ?, ?)",
            (name, phone, company))
        self.conn.commit()
        return cur.rowcount == 1          # False if the phone already exists

    def search(self, text: str) -> list[sqlite3.Row]:
        like = f"%{text}%"
        return self.conn.execute(
            "SELECT * FROM contacts WHERE name LIKE ? OR company LIKE ? "
            "ORDER BY name", (like, like)).fetchall()

    def update_company(self, phone: str, company: str) -> None:
        self.conn.execute("UPDATE contacts SET company = ? WHERE phone = ?",
                          (company, phone))
        self.conn.commit()

    def count(self) -> int:
        return self.conn.execute("SELECT COUNT(*) FROM contacts").fetchone()[0]


book = ContactBook(Path("contacts.db"))
for name, phone, company in [
    ("Siti Aminah", "+60120000001", "MU Packaging"),
    ("Jason Lim", "+60120000002", "Codencode"),
    ("Priya Nair", "+60120000003", "Sunrise Realty"),
]:
    added = book.add(name, phone, company)
    print(f"{'added  ' if added else 'exists '} {name}")

book.update_company("+60120000002", "Codencode Academy")
print("\nSearch 'code':")
for c in book.search("code"):
    print(f"  {c['name']:<16} {c['phone']}  {c['company']}")
print("total contacts:", book.count())

A project layout every developer recognises

whatIn plain words

Real projects separate code (src/), tests (tests/), data (data/) and docs (README). Anyone who opens your repo knows where everything is, and tools like pytest find things automatically.

likeEveryday analogy

A well-run kitchen: ingredients in the store room, utensils on the rack, recipes on the wall. A new cook can start in five minutes.

rulesTheory: the key rules

  • src/<package>/: one job per module
  • tests/ mirrors src/
  • README.md, requirements.txt, .gitignore at the root
  • Databases and .env are never committed

scaffold.py
"""scaffold.py: create a standard Python project layout in one command."""
from pathlib import Path

NAME = "kopitiam"

FILES = {
    f"src/{NAME}/__init__.py": '"""Kopitiam order system."""\n',
    f"src/{NAME}/models.py": "# dataclasses / classes go here\n",
    f"src/{NAME}/repository.py": "# all SQL lives here\n",
    f"src/{NAME}/cli.py": "# argparse entry point\n",
    "tests/__init__.py": "",
    f"tests/test_models.py": "def test_placeholder():\n    assert True\n",
    "data/.gitkeep": "",
    "README.md": f"# {NAME.title()}\n\nOne sentence about what it does.\n",
    "requirements.txt": "pytest\nruff\n",
    ".gitignore": ".venv/\n__pycache__/\n*.db\n.env\n",
}

root = Path(NAME)
for rel, content in FILES.items():
    path = root / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    if not path.exists():                 # never overwrite your work
        path.write_text(content, encoding="utf-8")

for path in sorted(root.rglob("*")):
    depth = len(path.relative_to(root).parts) - 1
    print("    " * depth + path.name + ("/" if path.is_dir() else ""))

argparse: turn a script into a real command

whatIn plain words

argparse reads the words typed after your script name, checks them, converts types, and gives you --help for free. It's how tools like git and pip take commands.

likeEveryday analogy

An order form with fixed boxes and tick options. The cashier knows exactly where to look, and a wrong entry is caught at the counter.

rulesTheory: the key rules

  • Positional: add_argument("name")
  • Optional: add_argument("--times", type=int, default=1)
  • choices= limits the allowed values
  • Subcommands: add_subparsers() (add / list / done)
syntax at a glance
parser = argparse.ArgumentParser(description="…")
parser.add_argument("name")
parser.add_argument("--lang", choices=["en", "bm"])

real lifetodo.py: a terminal to-do list saved in SQLite

A complete small tool in one file: argparse subcommands, a SQLite table, and functions that each do one thing. Add tasks with due dates, list them by urgency, tick them off. The tasks survive restarts.

todo.py
"""todo.py: a to-do list you run from the terminal, saved in SQLite.

  python todo.py add "Pay electricity bill" --due 2026-10-25
  python todo.py list
  python todo.py done 1
"""
import argparse
import sqlite3
from pathlib import Path

DB = Path("todo.db")


def connect() -> sqlite3.Connection:
    conn = sqlite3.connect(DB)
    conn.execute("""CREATE TABLE IF NOT EXISTS tasks (
        id INTEGER PRIMARY KEY, title TEXT NOT NULL,
        due TEXT, done INTEGER DEFAULT 0)""")
    return conn


def add(title: str, due: str | None) -> None:
    with connect() as conn:
        cur = conn.execute("INSERT INTO tasks (title, due) VALUES (?, ?)",
                           (title, due))
    print(f"added #{cur.lastrowid}: {title}")


def show() -> None:
    with connect() as conn:
        rows = conn.execute("SELECT id, title, due, done FROM tasks "
                            "ORDER BY done, due IS NULL, due").fetchall()
    if not rows:
        print("nothing to do")
    for task_id, title, due, done in rows:
        mark = "x" if done else " "
        print(f"[{mark}] #{task_id:<3} {title:<26} {due or ''}")


def done(task_id: int) -> None:
    with connect() as conn:
        cur = conn.execute("UPDATE tasks SET done = 1 WHERE id = ?", (task_id,))
    print(f"done #{task_id}" if cur.rowcount else f"no task #{task_id}")


def main() -> None:
    parser = argparse.ArgumentParser(description="Tiny to-do list")
    sub = parser.add_subparsers(dest="cmd", required=True)
    p_add = sub.add_parser("add", help="add a task")
    p_add.add_argument("title")
    p_add.add_argument("--due", help="YYYY-MM-DD")
    sub.add_parser("list", help="show tasks")
    p_done = sub.add_parser("done", help="mark a task done")
    p_done.add_argument("id", type=int)

    args = parser.parse_args()
    if args.cmd == "add":
        add(args.title, args.due)
    elif args.cmd == "list":
        show()
    else:
        done(args.id)


if __name__ == "__main__":
    main()
